先放源码：mini_cpu.prism

前三集交齐了三件东西：语言的形状（第一集）、一颗模块的写法（第二集）、多颗模块的组装（第三集）——组装的结果就是这颗 mini_cpu。这一集不写新代码：把 prism 命令行一条一条走完，每一件工具都对着这一份源码说话。先把完整源码放这儿（全 97 行，第三集拼的那份，一字未改）——接下来每件工具的输出，都可以回到这里对上任意一行：

// =============================================================================
// mini_cpu.prism —— 第三集样例：用 design + connect + wire + expose 把三颗
// 独立模块拼成一颗"迷你数据通路"。
//
//   ticker  →（wire：+1）→  doubler  →  thresh
//     心跳         组合            组合       比较
//
// 每颗模块都可以单独 run；拼起来之后，一颗的心跳驱动下一颗的动作。
// 这不是完整的 CPU，但是 CPU 的形状：PC → 取指 → 译码 → 执行 → 判分支。
// =============================================================================

module ticker {
    in  en  : Signal<u1>
    out c   : Signal<u7>
    reg cnt : u7

    on clock {
        cnt' = en ? cnt + 1 : cnt
    }

    c = cnt
}

module doubler {
    in  x : Signal<u8>
    out y : Signal<u8>

    y = x + x
}

module thresh {
    in  x   : Signal<u8>
    out hit : Signal<u1>

    hit = (x >= 16) ? 1 : 0
}

@schedule ticker {
    bitwidth  : { data:7 }
    pipeline  : 1 stage
    reset     : async_low
    clock     : 100 MHz, domain = single
    resource  : none
    latency   : 1 cycle
}

@schedule doubler {
    bitwidth  : { data:8 }
    pipeline  : 1 stage
    reset     : async_low
    clock     : 100 MHz, domain = single
    resource  : none
    latency   : 1 cycle
}

@schedule thresh {
    bitwidth  : { data:8 }
    pipeline  : 1 stage
    reset     : async_low
    clock     : 100 MHz, domain = single
    resource  : none
    latency   : 1 cycle
}

design mini_cpu {
    t : ticker
    d : doubler
    h : thresh

    // 《wire》：写表达式，落到一个具名网线上，再 `->` 推给下游端口
    wire step : u8 = t.c + 1 -> d.x
    wire res  : u8 = d.y     -> h.x

    // 《expose》：把实例内部 reg 引到顶层做观测端口
    expose t.cnt as ticks
}

// 逐拍手算（观测点在边沿之后）：
//   ticks = 1,2,3,…   step = ticks+1   res = step*2   hit = res >= 16
run mini_cpu_check of mini_cpu {
    stim : {
        t.en, [ticks, step, res, h.hit] = {
            1, [ 1,  2,  4, 0],
            1, [ 2,  3,  6, 0],
            1, [ 3,  4,  8, 0],
            1, [ 4,  5, 10, 0],
            1, [ 5,  6, 12, 0],
            1, [ 6,  7, 14, 0],
            1, [ 7,  8, 16, 1],
            1, [ 8,  9, 18, 1],
            1, [ 9, 10, 20, 1],
            1, [10, 11, 22, 1],
        }
    }
    until : end of stim
    show  : { t.en, ticks, step, res, h.hit }
}

先过四道静态关：check / typecheck / keywords / dump-ast

工具的第一件事不是生成代码，是在交底之前把该看的看一遍。四道关，一道比一道细——从最便宜的跑起。

bash解析 + 可下降性

  • 什么时候用它：改完源码的第一条命令——只解析、不推导，最便宜；适合挂在编辑器和 CI 的最前面。

// 时序通路：ticker（reg + on clock -> always_ff）
// 从 .prism 源码 + @schedule 自动生成。
//
//   bitwidth : data=32
//   reset    : async_low
//   clock    : 100 MHz
`timescale 1ns/1ps

module ticker_rtl #(
  parameter int DW = 32
)(
  input  logic clk,
  input  logic rst_n,
  input  logic [1-1:0] en,
  output logic [7-1:0] c,
  output logic [7-1:0] xpose__cnt
);

  // 时序寄存器
  logic [7-1:0] cnt;

  // 时序更新（reset=async_low）
  always_ff @(posedge clk or negedge rst_n) begin
    if (!rst_n) begin
      cnt <= '0;
    end else begin
      cnt <= (en ? (cnt + 1) : cnt);   // cnt' = ...
    end
  end

  // 组合输出 / 局部 wire 赋值
  assign c = cnt;

  // expose 观测端口驱动
  assign xpose__cnt = cnt;

endmodule

// 顶层网表：design mini_cpu（3 子模块 + 2 connect + 2 design-level assign + 2 内部 wire + 2 声明 wire + 1 expose 观测）
// 子模块例化 + connect 连线 + clk/rst 分发。仅 Signal 端口（裸电平）。
`timescale 1ns/1ps

module mini_cpu_top (
  input  logic clk,
  input  logic rst_n,
  input  logic [1-1:0] t__en,
  output logic [1-1:0] h__hit,
  output logic [7-1:0] ticks
);

  logic  [7:0] step;
  logic  [7:0] res;

  // 内部连线（connect 生成）
  logic  [7-1:0] w__t__c;
  logic  [8-1:0] w__d__y;

  // expose 观测线（子模块输出端口引出，非层次引用）
  logic [7-1:0] w__xpose__t__cnt;

  // 子模块例化（clk/rst_n 分发）
  ticker_rtl t (
    .clk(clk),
    .rst_n(rst_n),
    .en(t__en),
    .c(w__t__c),
    .xpose__cnt(w__xpose__t__cnt)
  );

  doubler_rtl d (
    .clk(clk),
    .rst_n(rst_n),
    .x(step),
    .y(w__d__y)
  );

  thresh_rtl h (
    .clk(clk),
    .rst_n(rst_n),
    .x(res),
    .hit(h__hit)
  );

  // design-level 组合 assign
  assign step = (w__t__c + 1);
  assign res = w__d__y;

  // 观测 assign（expose，走子模块输出端口）
  assign ticks = w__xpose__t__cnt;   // expose

endmodule
  • 第三方一眼可读：没有自定义运行时、没有库——就是普通 SV：always_ff + assign + 例化。丢给 iverilog（下一节）、或丢给综合器（后面几日）都直接用。
  • 顶层端口表兑现第三集的"推导"：t__en 是没人接的输入、h__hit 是没人接的输出——顶层那几个孔就是这么来的。
  • expose 绕了一层：给 ticker_rtl 加一个 xpose__cnt 观测端口，顶层 assign ticks = w__xpose__t__cnt——跨层次的读被拆成"端口 + 线"（yosys 不认跨层次引用，绕这一下才不静默丢线）。
  • 中间线名 w__t__c / w__d__y 就是第二集讲过的 __ 段分隔符——段接段，可逆读回层级。

// ===== AUTO-GENERATED by Prism TLM backend (SystemC) — DO NOT EDIT =====
// run mini_cpu_check / design mini_cpu：3 个实例 + 2 条 connect + 2 根 wire
// 逐拍：写本拍激励 -> 拉一个上升沿（组合链由 delta cycle 自行安定）-> 采样。
// TSV 列序与取值口径 = golden 的 run 表，可逐字比对。
#include <systemc.h>
#include <vector>
#include <string>
#include <cstdio>

using VW = sc_dt::sc_bigint<512>;   // 求值统一宽字：承载 512 位定宽信号
using UVW = sc_dt::sc_biguint<512>;  // 无符号视图（移位量判越界用）
struct V4 {
  VW v; int s;         // s: 0=确定 1=高阻 z 2=未知 x
  V4() : v(0), s(0) {}
  V4(VW vv, int ss) : v(vv), s(ss) {}
  V4(int vv, int ss) : v(VW((int64_t)vv)), s(ss) {}
};
inline bool operator==(const V4& a, const V4& b) { return a.v == b.v && a.s == b.s; }
inline bool operator!=(const V4& a, const V4& b) { return !(a == b); }
static const char* v4str(const V4& x);   // 前向：__int128 十进制打处在下方
inline std::ostream& operator<<(std::ostream& os, const V4& x) {
  if (x.s == 1) return os << "z";
  if (x.s == 2) return os << "x";
  return os << v4str(x);
}
inline void sc_trace(sc_core::sc_trace_file*, const V4&, const std::string&) {}

// 拼接段的合成状态：全段悬空才是 z，掺 x 或掺实数据就是 x（golden 的
// `Z if all(v is Z) else X` 同款）。段里一个非确定值都没有就是 0——
// 拼接的段状态在生成期多半是运行时的读端口表达式，不能按字面猜。
[[maybe_unused]] static int seg_state(const std::vector<int>& ss) {
  int z = 0, det = 0;
  for (size_t i = 0; i < ss.size(); i++) {
    if (ss[i] == 2) return 2;
    if (ss[i] == 1) z++; else det++;
  }
  if (z == 0) return 0;
  return det == 0 ? 1 : 2;
}

// 观测列的文本：确定值打十进制，z/x 打字母——TSV 才能和 golden 的表逐字比。
// sc_bigint<N>::to_string(radix=10) 直接给十进制串，和 Python str(int) 同形。
static const char* v4str(const V4& x) {
  static std::string buf;
  if (x.s == 1) return "z";
  if (x.s == 2) return "x";
  buf = x.v.to_string();
  return buf.c_str();
}

// ---- module ticker（实例 t）----
// @schedule latency = 1 cycle：事务级通路的建模延迟；本文件是
// 逐拍通路，一拍一次事务步，latency 不参与波形（见文件头说明）。
SC_MODULE(ticker_t_tlm) {
 public:
  sc_in<bool> clk_in{"clk"};
  sc_in<V4> en{"en"};   // 1 位 unsigned
  sc_out<V4> c{"c"};   // 7 位 unsigned
  V4 __o_c{0, 0};
  V4 cnt = V4{(VW(VW(0)) & ((VW(1) << 7) - VW(1))), 0};

  void settle() {
    __o_c = V4{(VW(cnt.v) & ((VW(1) << 7) - VW(1))), cnt.s};
    c.write(__o_c);
  }

  void tick() {
    settle();      // 先按当前输入与状态安定（组合链的每一级都这样）
    if (!clk_in.event()) return;   // 只是输入变了：不推进状态
    // 时钟沿：clock 块 RHS 用旧状态 + 本拍输入
    V4 __n0 = V4{(VW((((en.read().v) != 0) ? (VW((cnt.v) + (((VW((int64_t)0x1LL)))))) : (cnt.v))) & ((VW(1) << 7) - VW(1))), ((en.read().s) ? 2 : (((en.read().v) != 0) ? (((cnt.s) ? 2 : 0)) : (cnt.s)))};
    // 统一应用（非阻塞：上面每个 RHS 都基于沿前环境）
    cnt = __n0;
    settle();      // 沿后重算——观测相看这一组值
  }

  SC_CTOR(ticker_t_tlm) {
    SC_METHOD(tick); sensitive << clk_in.pos();
    sensitive << en;
  }
};

// ---- module doubler（实例 d）----
// @schedule latency = 1 cycle：事务级通路的建模延迟；本文件是
// 逐拍通路，一拍一次事务步，latency 不参与波形（见文件头说明）。
SC_MODULE(doubler_d_tlm) {
 public:
  sc_in<bool> clk_in{"clk"};

  }

};

static std::string g_trace;

SC_MODULE(Top) {
 public:
  // 时钟由驱动线程逐拍手动拉：sc_clock 在 t=0 白送一个上升沿，
  // 整张表会比 golden 提前一拍（三方对撞抓到的第一个真 bug）。
  sc_signal<bool> clk_sig;
  sc_signal<V4> w__step;   // 8 位
  sc_signal<V4> w__res;   // 8 位
  sc_signal<V4> o__t__c;
  sc_signal<V4> o__d__y;
  sc_signal<V4> o__h__hit;
  sc_signal<V4> d__t__en;
  ticker_t_tlm t{"t"};
  doubler_d_tlm d{"d"};
  thresh_h_tlm h{"h"};

  SC_CTOR(Top) : clk_sig("clk_sig") {
    clk_sig.write(false);
    SC_THREAD(drv);
    t.clk_in(clk_sig);
    t.c(o__t__c);
    t.en(d__t__en);
    d.clk_in(clk_sig);
    d.x(w__step);
    d.y(o__d__y);
    h.clk_in(clk_sig);
    h.hit(o__h__hit);
    h.x(w__res);
    SC_METHOD(net); sensitive << o__d__y << o__t__c;
  }

  void net() {
    w__step.write(V4{(VW(VW((o__t__c.read().v) + (((VW((int64_t)0x1LL)))))) & ((VW(1) << 8) - VW(1))), ((o__t__c.read().s) ? 2 : 0)});
    w__res.write(V4{(VW(o__d__y.read().v) & ((VW(1) << 8) - VW(1))), o__d__y.read().s});
  }
  void drv() {
    // run clock=100 MHz, reset=1 cycles
    wait(10 * 1, SC_NS);
    for (int k = 0; k < 10; k++) {
      d__t__en.write(V4{STIM[k][0], 0});
      wait(5, SC_NS);
      clk_sig.write(true);  wait(5, SC_NS);
      clk_sig.write(false); wait(SC_ZERO_TIME);
      char buf[512];
      std::snprintf(buf, sizeof buf, "%d", k);
      std::string row(buf);
      std::snprintf(buf, sizeof buf, "\t%s", v4str(d__t__en.read()));
      row += buf;
      std::snprintf(buf, sizeof buf, "\t%s", v4str(t.cnt));
      row += buf;
      std::snprintf(buf, sizeof buf, "\t%s", v4str(w__step.read()));
      row += buf;
      std::snprintf(buf, sizeof buf, "\t%s", v4str(w__res.read()));
      row += buf;
      std::snprintf(buf, sizeof buf, "\t%s", v4str(o__h__hit.read()));
      row += buf;
      g_trace += row + "\n";
    }
    sc_stop();
  }
};

extern "C" const char* prism_tlm_trace() {
  if (!g_trace.empty()) return g_trace.c_str();
  g_trace = "cycle";
  g_trace += "\tt_en";
  g_trace += "\tticks";
  g_trace += "\tstep";
  g_trace += "\tres";
  g_trace += "\th_hit";
  g_trace += "\n";
  Top top{"top"};
  sc_start();
  return g_trace.c_str();
}
int sc_main(int argc, char** argv) { prism_tlm_trace(); return 0; }

对撞：把两条腿摆到同一张桌上

诚实边界

本集从头到尾只有工具，没有整机、没有外设、没有上板。SoC 与 OpenSBI 是后面几日的事——这一集的任务，只是把"你将用来造它们的那套东西"一次看全。

"看全"的意思是：出错时它拦你（check / typecheck）、语言有多少保留字它记账（keywords）、你想看它陪你看树（dump-ast）、要出货它出货（gen-rtl / gen-tlm）、跑没跑对三台执行者各跑一遍（run 三档）、最后摆到一张桌上对撞（cosim）、连自己缺什么工具都记账（deps）。
